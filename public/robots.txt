# RFC 9309 group precedence: a crawler obeys ONLY the single most specific
# User-agent group that matches it and ignores every other group. A named
# "Allow: /"-only group therefore ERASES the /api/ + /admin Disallows for that
# bot — so search and AI-answer crawlers (Googlebot, Bingbot's siblings, GPTBot,
# OAI-SearchBot, ChatGPT-User, ClaudeBot, Claude-SearchBot, Claude-User,
# PerplexityBot, Perplexity-User, Google-Extended, Applebot-Extended, Amazonbot,
# meta-externalagent, DuckDuckBot, YandexBot, Baiduspider, …) are deliberately
# NOT named here. They all fall through to the * group below and inherit the
# full rule set; they remain welcome to crawl and cite the site. Named groups
# exist only where policy genuinely differs, and each carries the full rule set.

User-agent: *
# Content-Signal (contentsignals.org / draft-romm-aipref-contentsignals) — search
# indexing + AI-answer input allowed, AI model training opted out. It is served as
# an HTTP response header instead of a robots.txt directive (see next.config.mjs)
# because RFC 9309 defines no Content-Signal field, and emitting it here trips
# strict robots.txt validators ("Invalid robots.txt format").
Allow: /
# Social/SEO crawlers must be able to fetch CMS-uploaded OG & cover images, which
# Payload serves at /api/media/file/... . Without this, "Disallow: /api/" blocks
# them, so WhatsApp / X / Facebook / LinkedIn link previews render with no image.
# Longest-match wins, so every other /api/ path stays blocked.
Allow: /api/media/
Disallow: /api/
Disallow: /admin
# Cloudflare-internal paths (e.g. /cdn-cgi/l/email-protection) are not site pages.
Disallow: /cdn-cgi/

# ── Bing (also serves Microsoft Copilot) ───────────────────────
# Kept as a named group ONLY because Bingbot honours Crawl-delay (Googlebot
# ignores it, and setting it in * correlated with reduced Google crawl
# frequency). Since this group REPLACES * for Bingbot rather than extending
# it, the full Allow/Disallow set is repeated here.
User-agent: Bingbot
Allow: /
Allow: /api/media/
Disallow: /api/
Disallow: /admin
Disallow: /cdn-cgi/
Crawl-delay: 1

# ── Training-only scrapers (no citation value — blocked) ────────
# CCBot (Common Crawl) feeds model training datasets without driving
# citations or referral traffic, so it stays blocked.
User-agent: CCBot
Disallow: /

User-agent: SemrushBot
Disallow: /

Disallow: /

Disallow: /
